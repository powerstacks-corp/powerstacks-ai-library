# D12: What software reaches end of support in the next 90 days?

Back to [quick answers](../quick-answers.md).

```dax
DEFINE
    VAR AsOf = MAX ( 'Authorization'[Refresh Time] )
    VAR Pop =
        CALCULATETABLE (
            VALUES ( Device[Defender Device ID] ),
            Device[Onboarding Status] = "Onboarded",
            NOT ( Device[Is Excluded] = TRUE () )
        )
    VAR Installs =
        FILTER (
            ADDCOLUMNS (
                FILTER ( 'Software Devices', 'Software Devices'[Is Active] = TRUE () ),
                "Status", RELATED ( Software[End Of Support Status] ),
                "EOSDate", RELATED ( Software[End Of Support Date] ),
                "Product", RELATED ( Software[Software Name] ),
                "Publisher", RELATED ( Software[Publisher] )
            ),
            'Software Devices'[Defender Device ID] IN Pop
                && NOT ISBLANK ( [Status] ) && [Status] <> "None"
                && [EOSDate] >= TODAY () && [EOSDate] <= TODAY () + 90
        )
    VAR Pairs =
        DISTINCT ( SELECTCOLUMNS ( Installs, "D", [EOSDate], "Product", [Product], "Publisher", [Publisher],
            "Dev", 'Software Devices'[Defender Device ID] ) )
    VAR Total = COUNTROWS ( DISTINCT ( SELECTCOLUMNS ( Pairs, "d", [Dev] ) ) ) + 0
    VAR ByDate = GROUPBY ( Pairs, [D], [Product], [Publisher], "N", SUMX ( CURRENTGROUP (), 1 ) )
    VAR TopRows = TOPN ( 10, ByDate, [D], ASC, [N], DESC, [Product], ASC, [Publisher], ASC )
EVALUATE
GENERATEALL (
    ROW ( "AsOf", FORMAT ( AsOf, "yyyy-mm-dd hh:nn" ), "DevicesAffected", FORMAT ( Total, "#,0" ), "Entries", FORMAT ( COUNTROWS ( ByDate ) + 0, "#,0" ) ),
    SELECTCOLUMNS ( TopRows,
        "EOSDate", FORMAT ( [D], "yyyy-mm-dd" ),
        "Product", [Product],
        "Publisher", [Publisher],
        "Devices", FORMAT ( [N], "#,0" ) )
)
ORDER BY [EOSDate], [Product]
```

**Answer:** {DevicesAffected} devices have software reaching end of support in the next 90 days ({Entries} product and date entries; first 10 shown) (data as of {AsOf})

**Each row:** {EOSDate}: {Product} ({Publisher}), {Devices} devices
