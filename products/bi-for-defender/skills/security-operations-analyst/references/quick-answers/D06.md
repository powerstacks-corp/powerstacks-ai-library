# D06: Which five devices have the most critical vulnerabilities?

Back to [quick answers](../quick-answers.md).

```dax
DEFINE
    VAR AsOf = MAX ( 'Authorization'[Refresh Time] )
    VAR Pop =
        CALCULATETABLE (
            VALUES ( Device[Defender Device ID] ),
            Device[Onboarding Status] = "Onboarded",
            NOT ( Device[Is Excluded] = TRUE () )
        )
    VAR V =
        FILTER (
            ADDCOLUMNS (
                FILTER (
                    'Software Vulnerabilities',
                    'Software Vulnerabilities'[Is Active] = TRUE () && RELATED ( Vulnerability[Severity Level] ) = "1 - Critical"
                ),
                "Dev", RELATED ( 'Software Devices'[Defender Device ID] ),
                "Cve", 'Software Vulnerabilities'[vulnerability.id]
            ),
            [Dev] IN Pop
        )
    VAR Pairs = DISTINCT ( SELECTCOLUMNS ( V, "Dev", [Dev], "Cve", [Cve] ) )
    VAR ByDev = GROUPBY ( Pairs, [Dev], "N", SUMX ( CURRENTGROUP (), 1 ) )
    VAR Named = ADDCOLUMNS ( ByDev, "Name", LOOKUPVALUE ( Device[Device Name], Device[Defender Device ID], [Dev] ) )
    VAR TopRows = TOPN ( 5, Named, [N], DESC, [Name], ASC, [Dev], ASC )
EVALUATE
GENERATEALL (
    ROW ( "AsOf", FORMAT ( AsOf, "yyyy-mm-dd hh:nn" ), "DevicesWithCritical", FORMAT ( COUNTROWS ( ByDev ) + 0, "#,0" ) ),
    SELECTCOLUMNS ( TopRows,
        "Rank", RANKX ( TopRows, [N] ),
        "Device", [Name],
        "CriticalCVEs", FORMAT ( [N], "#,0" ) )
)
ORDER BY [Rank], [Device]
```

**Answer:** {DevicesWithCritical} devices have an active Critical vulnerability; the five with the most distinct Critical CVEs are below (data as of {AsOf})

**Each row:** {Rank}. {Device}: {CriticalCVEs} Critical CVEs
