# D11: Which software has the most vulnerabilities?

Back to [quick answers](../quick-answers.md).

```dax
DEFINE
    VAR AsOf = MAX ( 'Authorization'[Refresh Time] )
    VAR Pop =
        CALCULATETABLE (
            VALUES ( Device[Defender Device ID] ),
            Device[Onboarding Status] = "Onboarded",
            NOT ( Device[Is Excluded] = TRUE () )
        )
    VAR V =
        FILTER (
            ADDCOLUMNS (
                FILTER (
                    'Software Vulnerabilities',
                    'Software Vulnerabilities'[Is Active] = TRUE ()
                ),
                "Dev", RELATED ( 'Software Devices'[Defender Device ID] ),
                "Product", RELATED ( Software[Software Name] ),
                "Publisher", RELATED ( Software[Publisher] )
            ),
            [Dev] IN Pop
        )
    VAR CvePairs =
        DISTINCT ( SELECTCOLUMNS ( V, "Product", [Product], "Publisher", [Publisher],
            "Cve", 'Software Vulnerabilities'[vulnerability.id] ) )
    VAR ByProd = GROUPBY ( CvePairs, [Product], [Publisher], "N", SUMX ( CURRENTGROUP (), 1 ) )
    VAR TopRows = TOPN ( 5, ByProd, [N], DESC, [Product], ASC, [Publisher], ASC )
EVALUATE
GENERATEALL (
    ROW ( "AsOf", FORMAT ( AsOf, "yyyy-mm-dd hh:nn" ) ),
    SELECTCOLUMNS ( TopRows,
        "Rank", RANKX ( TopRows, [N] ),
        "Product", [Product],
        "Publisher", [Publisher],
        "CVEs", FORMAT ( [N], "#,0" ),
        "Devices", VAR p = [Product] VAR q = [Publisher] RETURN
            FORMAT ( COUNTROWS ( DISTINCT ( SELECTCOLUMNS ( FILTER ( V, [Product] = p && [Publisher] = q ), "d", [Dev] ) ) ), "#,0" ) )
)
ORDER BY [Rank], [Product]
```

**Answer:** Software with the most distinct active vulnerabilities (CVEs) (data as of {AsOf})

**Each row:** {Rank}. {Product} ({Publisher}): {CVEs} CVEs across {Devices} devices
