# D10: Break down our active vulnerabilities by severity.

Back to [quick answers](../quick-answers.md).

```dax
DEFINE
    VAR AsOf = MAX ( 'Authorization'[Refresh Time] )
    VAR Pop =
        CALCULATETABLE (
            VALUES ( Device[Defender Device ID] ),
            Device[Onboarding Status] = "Onboarded",
            NOT ( Device[Is Excluded] = TRUE () )
        )
    VAR V =
        FILTER (
            ADDCOLUMNS (
                FILTER (
                    'Software Vulnerabilities',
                    'Software Vulnerabilities'[Is Active] = TRUE ()
                ),
                "Dev", RELATED ( 'Software Devices'[Defender Device ID] ),
                "Sev", RELATED ( Vulnerability[Severity Level] )
            ),
            [Dev] IN Pop
        )
    VAR DevPairs = DISTINCT ( SELECTCOLUMNS ( V, "Sev", [Sev], "Dev", [Dev] ) )
    VAR CvePairs = DISTINCT ( SELECTCOLUMNS ( V, "Sev", [Sev], "Cve", 'Software Vulnerabilities'[vulnerability.id] ) )
    VAR BySev = GROUPBY ( DevPairs, [Sev], "N", SUMX ( CURRENTGROUP (), 1 ) )
EVALUATE
GENERATEALL (
    ROW ( "AsOf", FORMAT ( AsOf, "yyyy-mm-dd hh:nn" ),
        "DevicesWithAny", FORMAT ( COUNTROWS ( DISTINCT ( SELECTCOLUMNS ( DevPairs, "d", [Dev] ) ) ) + 0, "#,0" ) ),
    SELECTCOLUMNS ( BySev,
        "Severity", [Sev],
        "CVEs", VAR s = [Sev] RETURN FORMAT ( COUNTROWS ( FILTER ( CvePairs, [Sev] = s ) ), "#,0" ),
        "Devices", FORMAT ( [N], "#,0" ) )
)
ORDER BY [Severity]
```

**Answer:** {DevicesWithAny} devices have at least one active vulnerability; by severity (a device can appear in several rows) (data as of {AsOf})

**Each row:** {Severity}: {CVEs} distinct CVEs on {Devices} devices
