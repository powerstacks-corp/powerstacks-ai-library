# D04: Which missing security update affects the most devices?

Back to [quick answers](../quick-answers.md).

```dax
DEFINE
    VAR AsOf = MAX ( 'Authorization'[Refresh Time] )
    VAR Pop =
        CALCULATETABLE (
            VALUES ( Device[Defender Device ID] ),
            Device[Onboarding Status] = "Onboarded",
            NOT ( Device[Is Excluded] = TRUE () )
        )
    VAR V =
        FILTER (
            ADDCOLUMNS (
                FILTER (
                    'Software Vulnerabilities',
                    'Software Vulnerabilities'[Is Active] = TRUE () && NOT ISBLANK ( 'Software Vulnerabilities'[Missing Security Update ID] )
                ),
                "Dev", RELATED ( 'Software Devices'[Defender Device ID] )
            ),
            [Dev] IN Pop
        )
    VAR Pairs =
        DISTINCT ( SELECTCOLUMNS ( V, "KB", 'Software Vulnerabilities'[Missing Security Update ID], "Dev", [Dev] ) )
    VAR ByKb = GROUPBY ( Pairs, [KB], "N", SUMX ( CURRENTGROUP (), 1 ) )
    VAR TopRows = TOPN ( 5, ByKb, [N], DESC, [KB], ASC )
EVALUATE
GENERATEALL (
    ROW ( "AsOf", FORMAT ( AsOf, "yyyy-mm-dd hh:nn" ), "DistinctKBs", FORMAT ( COUNTROWS ( ByKb ) + 0, "#,0" ) ),
    SELECTCOLUMNS ( TopRows,
        "Rank", RANKX ( TopRows, [N] ),
        "KB", [KB],
        "Release", VAR k = [KB] RETURN
            MINX ( FILTER ( V, 'Software Vulnerabilities'[Missing Security Update ID] = k ),
                'Software Vulnerabilities'[Missing Security Update] ),
        "Devices", FORMAT ( [N], "#,0" ) )
)
ORDER BY [Rank], [KB]
```

**Answer:** {DistinctKBs} Microsoft security updates (KB numbers) are listed as missing; the ones missing on the most devices are below (data as of {AsOf})

**Each row:** {Rank}. KB{KB} ({Release}): missing on {Devices} devices
