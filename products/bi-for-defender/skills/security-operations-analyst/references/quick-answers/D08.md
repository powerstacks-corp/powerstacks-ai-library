# D08: What software is at or past end of support?

Back to [quick answers](../quick-answers.md).

```dax
DEFINE
    VAR AsOf = MAX ( 'Authorization'[Refresh Time] )
    VAR Pop =
        CALCULATETABLE (
            VALUES ( Device[Defender Device ID] ),
            Device[Onboarding Status] = "Onboarded",
            NOT ( Device[Is Excluded] = TRUE () )
        )
    VAR Installs =
        FILTER (
            ADDCOLUMNS (
                FILTER ( 'Software Devices', 'Software Devices'[Is Active] = TRUE () ),
                "Status", RELATED ( Software[End Of Support Status] ),
                "EOSDate", RELATED ( Software[End Of Support Date] ),
                "Product", RELATED ( Software[Software Name] ),
                "Publisher", RELATED ( Software[Publisher] )
            ),
            'Software Devices'[Defender Device ID] IN Pop
                && ( [Status] IN { "EOS Version", "EOS Software" }
                    || ( [Status] = "Upcoming EOS Version" && [EOSDate] < TODAY () ) )
        )
    VAR Pairs =
        DISTINCT ( SELECTCOLUMNS ( Installs, "Product", [Product], "Publisher", [Publisher],
            "Dev", 'Software Devices'[Defender Device ID] ) )
    VAR Total = COUNTROWS ( DISTINCT ( SELECTCOLUMNS ( Pairs, "d", [Dev] ) ) ) + 0
    VAR ByProd = GROUPBY ( Pairs, [Product], [Publisher], "N", SUMX ( CURRENTGROUP (), 1 ) )
    VAR TopRows = TOPN ( 5, ByProd, [N], DESC, [Product], ASC, [Publisher], ASC )
EVALUATE
GENERATEALL (
    ROW ( "AsOf", FORMAT ( AsOf, "yyyy-mm-dd hh:nn" ), "DevicesAffected", FORMAT ( Total, "#,0" ) ),
    SELECTCOLUMNS ( TopRows,
        "Rank", RANKX ( TopRows, [N] ),
        "Product", [Product],
        "Publisher", [Publisher],
        "Devices", FORMAT ( [N], "#,0" ) )
)
ORDER BY [Rank], [Product]
```

**Answer:** {DevicesAffected} devices run software at or past its end-of-support date; the most common products are below (data as of {AsOf})

**Each row:** {Rank}. {Product} ({Publisher}): {Devices} devices
