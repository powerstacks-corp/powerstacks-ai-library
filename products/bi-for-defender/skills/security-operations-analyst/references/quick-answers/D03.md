# D03: What is the most widespread exploitable vulnerability?

Back to [quick answers](../quick-answers.md).

```dax
DEFINE
    VAR AsOf = MAX ( 'Authorization'[Refresh Time] )
    VAR Pop =
        CALCULATETABLE (
            VALUES ( Device[Defender Device ID] ),
            Device[Onboarding Status] = "Onboarded",
            NOT ( Device[Is Excluded] = TRUE () )
        )
    VAR V =
        FILTER (
            ADDCOLUMNS (
                FILTER (
                    'Software Vulnerabilities',
                    'Software Vulnerabilities'[Is Active] = TRUE () && RELATED ( Vulnerability[Exploitability Level] ) = "Exploitable"
                ),
                "Dev", RELATED ( 'Software Devices'[Defender Device ID] ),
                "Cve", 'Software Vulnerabilities'[vulnerability.id]
            ),
            [Dev] IN Pop
        )
    VAR Pairs = DISTINCT ( SELECTCOLUMNS ( V, "CVE", [Cve], "Dev", [Dev] ) )
    VAR ByCve = GROUPBY ( Pairs, [CVE], "N", SUMX ( CURRENTGROUP (), 1 ) )
    VAR TopRows = TOPN ( 5, ByCve, [N], DESC, [CVE], ASC )
EVALUATE
GENERATEALL (
    ROW ( "AsOf", FORMAT ( AsOf, "yyyy-mm-dd hh:nn" ), "ExploitableCVEs", FORMAT ( COUNTROWS ( ByCve ) + 0, "#,0" ) ),
    SELECTCOLUMNS ( TopRows,
        "Rank", RANKX ( TopRows, [N] ),
        "CVE", [CVE],
        "Severity", LOOKUPVALUE ( Vulnerability[Severity Level], Vulnerability[vulnerability.id], [CVE] ),
        "CVSS", FORMAT ( LOOKUPVALUE ( Vulnerability[CVSS V3], Vulnerability[vulnerability.id], [CVE] ), "0.0" ),
        "Devices", FORMAT ( [N], "#,0" ) )
)
ORDER BY [Rank], [CVE]
```

**Answer:** {ExploitableCVEs} exploitable vulnerabilities are active; the most widespread by devices affected are listed below (data as of {AsOf})

**Each row:** {Rank}. {CVE} ({Severity}, CVSS {CVSS}): {Devices} devices
